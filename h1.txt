import { useCallback, useMemo, useState } from 'react';
import { ConfirmDialog } from '@/components/common/ConfirmDialog'
import { HistoryHeader } from '@/features/history/components/HistoryHeader';
import { ListPagination } from '@/components/common/ListPagination'
import { HistoryStatsGrid } from '@/features/history/components/HistoryStatsGrid';
import { HistoryTable } from '@/features/history/components/HistoryTable';
import { HistoryTimeline } from '@/features/history/components/HistoryTimeline';
import { HistoryToolbar } from '@/features/history/components/HistoryToolbar';
import { useHistoryEntries } from '@/features/history/hooks/useHistoryEntries';
import { buildHistoryStats } from '@/features/history/buildHistory';
import {
  DEFAULT_FILTERS,
  countByKind,
  filterHistory,
  filterWithoutKind,
} from '@/features/history/filterHistory';
import { paginate } from '@/features/history/paginate';
import type { HistoryEntry, HistoryFilters, HistoryView } from '@/features/history/types';
import { usePromptStore } from '@/stores/promptStore';

type PendingAction = { type: 'restore' | 'delete'; entry: HistoryEntry };

export default function HistoryPage() {
  const entries = useHistoryEntries();
  const restoreVersion = usePromptStore((s) => s.restoreVersion);
  const deleteVersion = usePromptStore((s) => s.deleteVersion);

  const [filters, setFilters] = useState<HistoryFilters>(DEFAULT_FILTERS);
  const [view, setView] = useState<HistoryView>('timeline');
  const [page, setPage] = useState(1);
  const [pending, setPending] = useState<PendingAction | null>(null);
  const [notice, setNotice] = useState<string | null>(null);
  const [now] = useState(() => new Date().getTime());

  const handleChange = useCallback((patch: Partial<HistoryFilters>) => {
    setFilters((prev) => ({ ...prev, ...patch }));
    setPage(1); // a new filter means the old page number no longer makes sense
  }, []);
  const handleReset = useCallback(() => {
    setFilters(DEFAULT_FILTERS);
    setPage(1);
  }, []);

  const handleRestore = useCallback(
    (entry: HistoryEntry) => setPending({ type: 'restore', entry }),
    [],
  );
  const handleDelete = useCallback(
    (entry: HistoryEntry) => setPending({ type: 'delete', entry }),
    [],
  );
  const handleCancel = useCallback(() => setPending(null), []);

  function handleConfirm(): void {
    if (!pending) return;
    const { type, entry } = pending;

    if (type === 'restore') {
      const created = restoreVersion(entry.promptId, entry.id);
      setNotice(
        created
          ? `Restored ${entry.label} of "${entry.promptTitle}" as new version ${created.label}.`
          : 'Could not restore this version.',
      );
    } else {
      const ok = deleteVersion(entry.promptId, entry.id);
      setNotice(
        ok
          ? `Deleted ${entry.label} of "${entry.promptTitle}".`
          : 'This version cannot be deleted.',
      );
    }
    setPending(null);
  }

  // KPIs describe the whole history, not the filtered view.
 const stats = useMemo(() => buildHistoryStats(entries, now), [entries, now]);

  const promptOptions = useMemo(() => {
    const seen = new Map<string, string>();
    for (const e of entries) seen.set(e.promptId, e.promptTitle);
    return [...seen]
      .map(([id, title]) => ({ id, title }))
      .sort((a, b) => a.title.localeCompare(b.title));
  }, [entries]);

  const { visible, kindCounts } = useMemo(() => {
  return {
    visible: filterHistory(entries, filters, now),
    kindCounts: countByKind(filterWithoutKind(entries, filters, now)),
  };
}, [entries, filters, now]);

  const pageResult = useMemo(() => paginate(visible, page), [visible, page]);

  return (
    <div className="space-y-6">
      <HistoryHeader entries={entries} />
      <HistoryStatsGrid stats={stats} />

      <HistoryToolbar
        filters={filters}
        onChange={handleChange}
        onReset={handleReset}
        promptOptions={promptOptions}
        kindCounts={kindCounts}
        view={view}
        onViewChange={setView}
      />

      {notice && (
        <p
          role="status"
          className="rounded-lg border border-primary/30 bg-primary/10 px-4 py-2 text-sm text-primary"
        >
          {notice}
        </p>
      )}

      {visible.length === 0 ? (
        <div className="rounded-xl border border-dashed border-border p-10 text-center text-sm text-muted-foreground">
          {entries.length === 0
            ? 'No versions saved yet. Save a version from the editor and it will appear here.'
            : 'No version logs match your filters.'}
        </div>
      ) : (
        <>
          {view === 'timeline' ? (
            <HistoryTimeline entries={pageResult.items} onRestore={handleRestore} onDelete={handleDelete} />
          ) : (
            <HistoryTable entries={pageResult.items} onRestore={handleRestore} onDelete={handleDelete} />
          )}
          <ListPagination result={pageResult} onPageChange={setPage} />
        </>
      )}

      {pending && (
        <ConfirmDialog
          tone={pending.type === 'delete' ? 'danger' : 'primary'}
          title={
            pending.type === 'restore'
              ? `Restore ${pending.entry.label}?`
              : `Delete ${pending.entry.label}?`
          }
          description={
            pending.type === 'restore'
              ? `"${pending.entry.promptTitle}" will get a new version with the content of ${pending.entry.label}. Your existing history is kept, nothing is overwritten.`
              : `${pending.entry.label} of "${pending.entry.promptTitle}" will be permanently removed from its history. The current version is not affected.`
          }
          confirmLabel={pending.type === 'restore' ? 'Restore version' : 'Delete version'}
          onConfirm={handleConfirm}
          onCancel={handleCancel}
        />
      )}
    </div>
  );
}import { Check, Copy } from 'lucide-react';
import { useEffect, useRef, useState } from 'react';

export function CommitHash({ hash }: { hash: string }) {
  const [copied, setCopied] = useState(false);
  const timer = useRef<number | undefined>(undefined);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  async function handleCopy(): Promise<void> {
    try {
      await navigator.clipboard.writeText(hash);
      setCopied(true);
      window.clearTimeout(timer.current);
      timer.current = window.setTimeout(() => setCopied(false), 1500);
    } catch {
      /* clipboard blocked: ignore silently */
    }
  }

  return (
    <button
      type="button"
      onClick={handleCopy}
      aria-label={copied ? 'Commit hash copied' : `Copy commit hash ${hash}`}
      className="inline-flex items-center gap-1.5 rounded-md bg-muted px-2 py-1 font-mono text-xs text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-2 focus-visible:outline-primary"
    >
      #{hash}
      {copied ? <Check className="size-3 text-emerald-500" aria-hidden="true" /> : <Copy className="size-3" aria-hidden="true" />}
    </button>
  );
}import { Download } from 'lucide-react';
import type { HistoryEntry } from '../types';

interface HistoryHeaderProps {
  entries: HistoryEntry[];
}

function exportAuditLog(entries: HistoryEntry[]): void {
  const blob = new Blob([JSON.stringify(entries, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `promptlab-audit-log-${new Date().toISOString().slice(0, 10)}.json`;
  a.click();
  URL.revokeObjectURL(url);
}

export function HistoryHeader({ entries }: HistoryHeaderProps) {
  return (
    <header className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
      <div className="space-y-2">
        <span className="inline-flex items-center gap-2 rounded-full border border-primary/30 bg-primary/10 px-3 py-1 text-xs font-medium text-primary">
          <span className="relative flex size-2">
            <span className="absolute inline-flex size-full animate-ping rounded-full bg-primary opacity-60" />
            <span className="relative inline-flex size-2 rounded-full bg-primary" />
          </span>
          {entries.length} Audited Revisions
        </span>
        <h1 className="font-heading text-3xl font-bold tracking-tight">History &amp; Version Logs</h1>
        <p className="max-w-xl text-sm text-muted-foreground">
          Every saved version of every prompt, with line and token changes. Restoring never erases
          history, it adds a new version.
        </p>
      </div>

      <button
        type="button"
        onClick={() => exportAuditLog(entries)}

        className="inline-flex items-center gap-2 rounded-lg border border-border bg-card px-4 py-2 text-sm font-medium transition-colors hover:bg-muted focus-visible:outline-2 focus-visible:outline-primary disabled:cursor-not-allowed disabled:opacity-50"

import gsap from 'gsap';
import { GitCommitHorizontal, RotateCcw, Layers, Activity } from 'lucide-react';
import type { LucideIcon } from 'lucide-react';
import type { HistoryStats } from '../types';

interface HistoryStatsGridProps {
  stats: HistoryStats;
}

interface KpiCard {
  key: string;
  label: string;
  value: string;
  hint: string | null;
  icon: LucideIcon;
  tone: 'primary' | 'danger' | 'secondary' | 'accent';
}

const toneClasses: Record<KpiCard['tone'], string> = {
  primary: 'bg-primary/10 text-primary',
  danger: 'bg-red-500/10 text-red-500',
  secondary: 'bg-violet-500/10 text-violet-500',
  accent: 'bg-pink-500/10 text-pink-500',
};

function buildCards(stats: HistoryStats): KpiCard[] {
  const trend =
    stats.commitsTrend === null
      ? null
      : `${stats.commitsTrend >= 0 ? '+' : ''}${stats.commitsTrend}% vs previous 30 days`;

  return [
    { key: 'commits', label: 'Total Commits', value: String(stats.totalCommits), hint: trend, icon: GitCommitHorizontal, tone: 'primary' },
    { key: 'rollbacks', label: 'Rollbacks Triggered', value: String(stats.rollbacks), hint: null, icon: RotateCcw, tone: 'danger' },
    { key: 'prompts', label: 'Prompts Versioned', value: String(stats.promptsVersioned), hint: null, icon: Layers, tone: 'secondary' },
    { key: 'daily', label: 'Avg. Daily Revisions', value: String(stats.avgDaily), hint: 'last 30 days', icon: Activity, tone: 'accent' },
  ];
}

export function HistoryStatsGrid({ stats }: HistoryStatsGridProps) {
  const rootRef = useRef<HTMLDivElement>(null);
  const cards = buildCards(stats);

  useLayoutEffect(() => {
    const root = rootRef.current;
    if (!root) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

    const ctx = gsap.context(() => {
      gsap.fromTo(
        '[data-kpi]',
        { y: 16, opacity: 0 },
        { y: 0, opacity: 1, duration: 0.5, stagger: 0.08, ease: 'power2.out', clearProps: 'transform,opacity' },
      );
    }, root);

    return () => ctx.revert();
  }, []);

  return (
    <div ref={rootRef} className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
      {cards.map(({ key, label, value, hint, icon: Icon, tone }) => (
        <section key={key} data-kpi className="rounded-xl border border-border bg-card p-5">
          <div className="flex items-start justify-between">
            <p className="text-sm text-muted-foreground">{label}</p>
            <span className={`flex size-9 items-center justify-center rounded-lg ${toneClasses[tone]}`}>
              <Icon className="size-4" aria-hidden="true" />
            </span>
          </div>
          <p className="mt-3 font-heading text-3xl font-bold">{value}</p>
          {hint && <p className="mt-1 text-xs text-muted-foreground">{hint}</p>}
        </section>
      ))}
    </div>
  );
}import { formatDistanceToNow, format } from 'date-fns';
import { ExternalLink, Eye, EyeOff, GitCompareArrows, RotateCcw, Trash2 } from 'lucide-react';
import { useId, useMemo, useState } from 'react';
import { Link } from 'react-router-dom';
import { diffLines } from '@/features/versions/diff';
import type { HistoryEntry } from '../types';
import { CommitHash } from './CommitHash';

interface HistoryLogCardProps {
  entry: HistoryEntry;
  onRestore: (entry: HistoryEntry) => void;
  onDelete: (entry: HistoryEntry) => void;
}

const lineClass = {
  added: 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400',
  removed: 'bg-red-500/10 text-red-600 line-through dark:text-red-400',
  same: 'text-muted-foreground',
} as const;

const linePrefix = { added: '+ ', removed: '− ', same: '  ' } as const;

const actionBtn =
  'inline-flex items-center gap-1.5 rounded-lg border border-border px-3 py-1.5 text-xs font-medium transition-colors hover:bg-muted focus-visible:outline-2 focus-visible:outline-primary disabled:cursor-not-allowed disabled:opacity-40 disabled:hover:bg-transparent';

export function HistoryLogCard({ entry, onRestore, onDelete }: HistoryLogCardProps) {
  const [showDiff, setShowDiff] = useState(false);
  const diffId = useId();

  const diff = useMemo(
    () => (showDiff ? diffLines(entry.previousContent ?? '', entry.content) : []),
    [showDiff, entry.previousContent, entry.content],
  );

  const date = new Date(entry.createdAt);
  const tokenText = `${entry.tokenDelta >= 0 ? '+' : ''}${entry.tokenDelta} tokens`;

  return (
    <article
      className={`rounded-xl border bg-card p-5 ${
        entry.isRollback ? 'border-red-500/40' : 'border-border'
      }`}
    >
      {/* Title row */}
      <div className="flex flex-wrap items-center gap-x-3 gap-y-2">
        <Link
          to={`/editor/${entry.promptId}`}
          className="font-heading text-base font-semibold hover:text-primary focus-visible:outline-2 focus-visible:outline-primary"
        >
          {entry.promptTitle}
        </Link>
        <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-medium text-primary">
          {entry.label}
        </span>
        {entry.isCurrent && (
          <span className="rounded-full bg-emerald-500/10 px-2 py-0.5 text-[11px] font-semibold uppercase tracking-wide text-emerald-600 dark:text-emerald-400">
            Current
          </span>
        )}
        {entry.isRollback && (
          <span className="inline-flex items-center gap-1 rounded-full bg-red-500/10 px-2 py-0.5 text-[11px] font-semibold uppercase tracking-wide text-red-500">
            <RotateCcw className="size-3" aria-hidden="true" />
            Rollback
          </span>
        )}
        <span className="ms-auto">
          <CommitHash hash={entry.hash} />
        </span>
      </div>

      {/* Author + time */}
      <p className="mt-2 flex items-center gap-2 text-xs text-muted-foreground">
        <span className="flex size-5 items-center justify-center rounded-full bg-primary/15 text-[10px] font-bold text-primary" aria-hidden="true">
          M
        </span>
        Maha
        <span aria-hidden="true">·</span>
        <time dateTime={entry.createdAt} title={format(date, 'PPpp')}>
          {formatDistanceToNow(date, { addSuffix: true })}
        </time>
      </p>

      {/* Note */}
      <p className="mt-3 rounded-lg bg-muted/60 px-3 py-2 text-sm">{entry.note || 'No description'}</p>

      {/* Telemetry chips */}
      <ul className="mt-3 flex flex-wrap gap-2 text-xs font-medium" aria-label="Changes in this version">
        <li className="rounded-md bg-emerald-500/10 px-2 py-1 text-emerald-600 dark:text-emerald-400">
          +{entry.linesAdded} lines
        </li>
        <li className="rounded-md bg-red-500/10 px-2 py-1 text-red-600 dark:text-red-400">
          −{entry.linesRemoved} lines
        </li>
        <li className="rounded-md bg-muted px-2 py-1 text-muted-foreground">{tokenText}</li>
        {entry.restoredFrom && (
          <li className="rounded-md bg-muted px-2 py-1 text-muted-foreground">
            from {entry.restoredFrom}
          </li>
        )}
      </ul>

      {/* Actions */}
      <div className="mt-4 flex flex-wrap gap-2">
        <button
          type="button"
          className={actionBtn}
          aria-expanded={showDiff}
          aria-controls={diffId}
          onClick={() => setShowDiff((v) => !v)}
        >
          {showDiff ? <EyeOff className="size-3.5" aria-hidden="true" /> : <Eye className="size-3.5" aria-hidden="true" />}
          {showDiff ? 'Hide Diff' : 'View Diff'}
        </button>

        <button
          type="button"
          className={actionBtn}
          onClick={() => onRestore(entry)}
          disabled={entry.isCurrent}
          title={entry.isCurrent ? 'This is already the current version' : undefined}
        >
          <RotateCcw className="size-3.5" aria-hidden="true" />
          Restore This Version
        </button>

        <Link to={`/editor/${entry.promptId}`} className={actionBtn}>
          <ExternalLink className="size-3.5" aria-hidden="true" />
          Open in Editor
        </Link>
        {!entry.isInitial && (
  <Link to={`/compare?prompt=${entry.promptId}&target=${entry.id}`} className={actionBtn}>
    <GitCompareArrows className="size-3.5" aria-hidden="true" />
    Compare
  </Link>
)}
        <button
          type="button"
          className={`${actionBtn} ms-auto text-red-500`}
          onClick={() => onDelete(entry)}
          disabled={entry.isCurrent}
          aria-label={`Delete version ${entry.label}`}
          title={entry.isCurrent ? 'The current version cannot be deleted' : undefined}
        >
          <Trash2 className="size-3.5" aria-hidden="true" />
        </button>
      </div>

      {/* Inline diff */}
      {showDiff && (
        <pre
          id={diffId}
          data-lenis-prevent
          className="mt-3 max-h-64 overflow-auto rounded-lg border border-border bg-background p-3 font-mono text-xs leading-relaxed"
        >
          {diff.map((line, i) => (
            <div key={i} className={`whitespace-pre-wrap px-1 ${lineClass[line.type]}`}>
              {linePrefix[line.type]}
              {line.text || ' '}
            </div>
          ))}
        </pre>
      )}
    </article>
  );
}
